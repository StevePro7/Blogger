&nbsp;&nbsp;class ServiceRegistry:
&nbsp;&nbsp;    _services: dict = {}
&nbsp;&nbsp;
&nbsp;&nbsp;    @classmethod
&nbsp;&nbsp;    def register(cls, name: str, service):
&nbsp;&nbsp;        cls._services[name] = service
&nbsp;&nbsp;
&nbsp;&nbsp;    @classmethod
&nbsp;&nbsp;    def get(cls, name: str):
&nbsp;&nbsp;        return cls._services[name]
&nbsp;&nbsp;
&nbsp;&nbsp;    @classmethod
&nbsp;&nbsp;    def clear(cls):
&nbsp;&nbsp;        cls._services.clear()
