Builds game infrastructure and acts as an intermediary between the game bootstrap and the service registry which delegates game init, load, update and draw game loop methods to actual game implementation code.

&nbsp;&nbsp;class AnGame:
&nbsp;&nbsp;    def __init__(self):
&nbsp;&nbsp;        self.running: bool = True
&nbsp;&nbsp;        build_game()
&nbsp;&nbsp;
&nbsp;&nbsp;    def Run(self):
&nbsp;&nbsp;        self.Initialize()
&nbsp;&nbsp;        self.LoadContent()
&nbsp;&nbsp;
&nbsp;&nbsp;        while self.running:
&nbsp;&nbsp;            MyGame.Manager.EventManager.ProcessEvents()
&nbsp;&nbsp;            if MyGame.Manager.EventManager.QuitRequested:
&nbsp;&nbsp;                self.Exit()
&nbsp;&nbsp;
&nbsp;&nbsp;            deltaTime: int = MyGame.Manager.ClockManager.Update()
&nbsp;&nbsp;            self.Update(deltaTime)
&nbsp;&nbsp;            self.Draw()
