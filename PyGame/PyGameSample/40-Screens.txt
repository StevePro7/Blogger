&nbsp;&nbsp;class BaseScreen(ABC):
&nbsp;&nbsp;

&nbsp;&nbsp;        self.Timer: int = None
&nbsp;&nbsp;
&nbsp;&nbsp;    @abstractmethod
&nbsp;&nbsp;    def LoadContent(self) -&gt; None:
&nbsp;&nbsp;        self.Timer = 0
&nbsp;&nbsp;
&nbsp;&nbsp;    @abstractmethod
&nbsp;&nbsp;    def Update(self, deltaTime: int) -&gt; ScreenType | None:
&nbsp;&nbsp;        return None
&nbsp;&nbsp;
&nbsp;&nbsp;    @abstractmethod
&nbsp;&nbsp;    def Draw(self) -&gt; None:
&nbsp;&nbsp;        pass
